cmake_minimum_required(VERSION 3.20)
project(gffx_image_transfer_scalar_test LANGUAGES C)
enable_testing()
add_executable(image_transfer_scalar ../test_image_transfer_scalar.c)
add_library(image_transfer_probe SHARED ../test_image_transfer_scalar.c)
foreach(target image_transfer_scalar image_transfer_probe)
  target_include_directories(${target} PRIVATE ../../../native/core)
  set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  if(MSVC)
    target_compile_options(${target} PRIVATE /fp:strict /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -ffp-contract=off -fno-fast-math -UNDEBUG)
    target_link_libraries(${target} PRIVATE m)
  endif()
endforeach()
add_test(NAME image_transfer_scalar COMMAND image_transfer_scalar)
