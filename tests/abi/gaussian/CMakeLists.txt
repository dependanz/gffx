cmake_minimum_required(VERSION 3.20)
project(gffx_gaussian_scalar_tests LANGUAGES C)
enable_testing()
add_library(gaussian_probe SHARED ../test_gaussian_scalar.c)
add_executable(gaussian_scalar ../test_gaussian_scalar.c)
foreach(target gaussian_probe gaussian_scalar)
  target_include_directories(${target} PRIVATE ../../../native/core)
  set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
  if(MSVC)
    target_compile_options(${target} PRIVATE /fp:strict /UNDEBUG)
  else()
    target_compile_options(${target} PRIVATE -ffp-contract=off -fno-fast-math -UNDEBUG)
    target_link_libraries(${target} PRIVATE m)
  endif()
endforeach()
add_test(NAME gaussian.scalar COMMAND gaussian_scalar)
